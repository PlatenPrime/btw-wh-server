document.addEventListener('DOMContentLoaded', () => {
    const excludeSelectors = [
        'button[aria-label="Minus"]',
        'input[aria-label="quantity"]',
        'button[aria-label="plus"]',
        '.button-group',
        '.options-category--radio-block'
    ];

    let touchStartX = 0;
    let touchStartY = 0;

    const handleTouchStart = (event) => {
        touchStartX = event.touches[0].clientX;
        touchStartY = event.touches[0].clientY;
    };

    const handleTouchEnd = (event) => {
        const touchEndX = event.changedTouches[0].clientX;
        const touchEndY = event.changedTouches[0].clientY;

        const diffX = Math.abs(touchEndX - touchStartX);
        const diffY = Math.abs(touchEndY - touchStartY);

        if (diffX < 10 && diffY < 10) {
            handleClick(event);
        }
    };

    const handleClick = (event) => {
        let isExcluded = excludeSelectors.some(selector => event.target.closest(selector));

        if (!isExcluded) {
            const link = event.currentTarget.querySelector('.product-title > a');
            if (link) {
                if (event.type === 'click' && event.button === 1) {
                    // Middle mouse button click
                    window.open(link.href, '_blank');
                } else if (event.type === 'click' && event.button === 0) {
                    // Left mouse button click
                    link.click();
                }
            }
        }
    };

    const initializeProductCards = () => {
        document.querySelectorAll('.product-layout').forEach(card => {
            card.removeEventListener('click', handleClick);
            card.removeEventListener('auxclick', handleClick);
            card.removeEventListener('touchstart', handleTouchStart);
            card.removeEventListener('touchend', handleTouchEnd);

            card.addEventListener('click', handleClick);
            card.addEventListener('auxclick', handleClick); // To handle middle mouse button
            card.addEventListener('touchstart', handleTouchStart);
            card.addEventListener('touchend', handleTouchEnd);
        });
    };

    // Initial initialization
    initializeProductCards();

    // Observe changes in the DOM
    const observer = new MutationObserver(mutations => {
        mutations.forEach(mutation => {
            if (mutation.type === 'childList') {
                initializeProductCards();
            }
        });
    });

    observer.observe(document.body, {
        childList: true,
        subtree: true
    });
});
